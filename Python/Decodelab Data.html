# Convert PascalCase names to snake_case
import re

def pascal_to_snake(cols):
    result=[]
    for value in cols:
        value = re.sub(r'([A-Z]+)([A-Z][a-z])', r'\1_\2', value)
        value = re.sub(r'([a-z0-9])([A-Z])', r'\1_\2', value)
        value = value.lower()
        result.append(value)
    return result
